module Declarations {
  const N: int
  var x: int
  assume positive = N > 0
  pure def increment(a: int): int = a + 1
  pure val initial = 0
  val current = x
  def next(a) = a + 1
  action init = x' = 0
  run simulate = all { init }
  temporal invariant = always(x >= 0)
}
