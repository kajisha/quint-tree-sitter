module Blocks {
  val conjunction = and { a, b, }
  val disjunction = or { a, b }
  action both = all { a, b, }
  action either = any { a, b }
  val grouped = { a + b }
  val local = {
    val increment(x) = x + 1
    increment(1)
  }
}
