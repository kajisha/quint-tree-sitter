module Matching {
  type Option[a] = | Some(a) | None
  def unwrap(value) = match value {
    | Some(x) => x
    | None => 0
    | _ => -1
  }
  val lower = match value { present(item) => item | absent => 0 }
  val ignored = match value { Some(_) => 1 | _ => 0 }
  val incremented = match value { Some(x) => x | None => 0 } + 1
  val argument = f(match value { Some(x) => x | None => 0 }, 2)
}
